// placed is one component of the design: where it sits and how it is made.
type placed struct {
	Kind  string
	Name  string
	X, Y  int
	W, H  int
	Props map[string]string
	// Fit and MinW, MinH are set on a component that follows the window: Fit
	// gives its rectangle for a window of w by h cells.
	Fit func(w, h int) (x, y, cw, ch int)
	// FitEnv is Fit for a component of a screen whose place also depends on the
	// screen's inputs or on the components before it.
	FitEnv     func(e fitEnv) (x, y, cw, ch int)
	MinW, MinH int
	// Drag lets the person move the component, Resize lets them resize it by
	// its bottom-right corner. Dx, Dy, Dw and Dh are how far they moved it from
	// where Fit puts it, so a window resize keeps their change.
	Drag, Resize   bool
	Dx, Dy, Dw, Dh int
	// Member is set on a component placed in a container: Of is the container's
	// index in the layout and Page the page it belongs to. It is shown only
	// while that page is.
	Member bool
	Of     int
	Page   int
}

// fitEnv is what a layout expression of a screen reads besides the window: its
// number and yes/no inputs ($Count) and the places of the components before it
// (below("Name")).
type fitEnv struct {
	W, H int
	// Inputs gives an input by name; a yes/no input is 1 or 0.
	Inputs func(name string) float64
	rects  map[string][4]int
}

// In is the input called name, 0 when there is none.
func (e fitEnv) In(name string) float64 {
	if e.Inputs == nil {
		return 0
	}
	return e.Inputs(name)
}

// Rect is one edge of the component called name: x, y, w, h, right or bottom.
func (e fitEnv) Rect(name, field string) float64 {
	r, ok := e.rects[name]
	if !ok {
		return 0
	}
	switch field {
	case "x":
		return float64(r[0])
	case "y":
		return float64(r[1])
	case "w":
		return float64(r[2])
	case "h":
		return float64(r[3])
	case "right":
		return float64(r[0] + r[2])
	case "bottom":
		return float64(r[1] + r[3])
	}
	return 0
}

// cells rounds a calculated size down to whole cells.
func cells(v float64) int {
	if math.IsNaN(v) || math.IsInf(v, 0) {
		return 0
	}
	return int(math.Floor(v + 1e-9))
}

// div is a division that gives 0 instead of infinity for a zero divisor.
func div(a, b float64) float64 {
	if b == 0 {
		return 0
	}
	return a / b
}

// part is a running component.
type part struct {
	update func(tea.Msg) tea.Cmd
	view   func() string
	// focus gives or takes the keyboard; nil when there is nothing to do.
	focus func(on bool) tea.Cmd
	start tea.Cmd
	// keys is true when the component reads the keyboard.
	keys bool
}

type props map[string]string

func (p props) flag(key string) bool { return p[key] == "true" }

func (p props) integer(key string, fallback int) int {
	n, err := strconv.Atoi(strings.TrimSpace(p[key]))
	if err != nil {
		return fallback
	}
	return n
}

func orDefault(v, fallback string) string {
	if v == "" {
		return fallback
	}
	return v
}

func fg(c string) lipgloss.Style {
	if c == "" {
		return lipgloss.NewStyle()
	}
	return lipgloss.NewStyle().Foreground(lipgloss.Color(c))
}

// split cuts s at sep and trims the pieces. A backslash before a comma, a
// semicolon or another backslash is that character itself, so an item can hold
// one.
func split(s, sep string) []string { return splitEscaped(s, sep, true) }

// splitRaw is split that keeps the backslashes, for the first cut of a table:
// the rows are cut at semicolons, then each row at commas.
func splitRaw(s, sep string) []string { return splitEscaped(s, sep, false) }

func splitEscaped(s, sep string, unescape bool) []string {
	var out []string
	var cur strings.Builder
	flush := func() {
		if piece := strings.TrimSpace(cur.String()); piece != "" {
			out = append(out, piece)
		}
		cur.Reset()
	}
	for i := 0; i < len(s); i++ {
		c := s[i]
		if c == '\\' && i+1 < len(s) && (s[i+1] == ',' || s[i+1] == ';' || s[i+1] == '\\') {
			if !unescape {
				cur.WriteByte(c)
			}
			cur.WriteByte(s[i+1])
			i++
			continue
		}
		if strings.HasPrefix(s[i:], sep) {
			flush()
			i += len(sep) - 1
			continue
		}
		cur.WriteByte(c)
	}
	flush()
	return out
}

// extra holds the components that need a file and a module of their own
// (bigtext.go, qrcode.go, image.go); each file registers itself here.
var extra = map[string]func(c placed, p props) *part{}

// build makes the running component for one placed component, cut to the
// rectangle it was given: a Bubbles help line or a long title never spills over
// its neighbours.
func build(c placed) *part {
	pt := buildPart(c)
	view := pt.view
	pt.view = func() string {
		return lipgloss.NewStyle().MaxWidth(c.W).MaxHeight(c.H).Render(view())
	}
	return pt
}

func buildPart(c placed) *part {
	p := props(c.Props)
	switch c.Kind {
	case "lipgloss.box":
		return staticPart(boxOf(c.W, c.H, p))
	case "lipgloss.label":
		return staticPart(label(c.W, p))
	case "lipgloss.list":
		return staticPart(listBlock(c.W, c.H, p))
	case "lipgloss.swatch":
		return staticPart(swatchBlock(c.H, p))
	case "bubbles.tree":
		return treePart(p, c.W, c.H)
	case "huh.spinner":
		return spinnerPart(props{"style": p.str("style"), "color": p.str("color"), "label": p.str("title")})
	case "community.frame":
		return staticPart(frame(c.W, c.H, p))
	case "community.dialog":
		return staticPart(dialog(c.W, c.H, p))
	case "community.statusmessage":
		return statusMessagePart(p, c.W)
	case "community.toast":
		return staticPart(toast(c.W, c.H, p))
	case "bubbles.textinput":
		return textInputPart(p, c.W)
	case "bubbles.textarea":
		return textAreaPart(p, c.W, c.H)
	case "bubbles.list":
		return listPart(p, c.W, c.H)
	case "bubbles.table":
		return tablePart(p, c.W, c.H)
	case "bubbles.viewport":
		return viewportPart(p, c.W, c.H)
	case "bubbles.paginator":
		return paginatorPart(p)
	case "bubbles.spinner":
		return spinnerPart(p)
	case "bubbles.progress":
		return progressPart(p, c.W)
	case "bubbles.stopwatch":
		return stopwatchPart(p)
	case "bubbles.timer":
		return timerPart(p)
	}
	if make, ok := extra[c.Kind]; ok {
		return make(c, p)
	}
	return staticPart(box(c.W, c.H, "normal", "240", "", ""))
}

func (p props) str(key string) string { return p[key] }

func staticPart(content string) *part {
	return &part{update: func(tea.Msg) tea.Cmd { return nil }, view: func() string { return content }}
}

func borderOf(name string) lipgloss.Border {
	switch name {
	case "normal":
		return lipgloss.NormalBorder()
	case "thick":
		return lipgloss.ThickBorder()
	case "double":
		return lipgloss.DoubleBorder()
	case "ascii":
		return lipgloss.ASCIIBorder()
	case "hidden":
		return lipgloss.HiddenBorder()
	}
	return lipgloss.RoundedBorder()
}

// box draws a w by h frame with an optional title in the top edge. A gradient
// colour blends the border from color at the left edge to it at the right.
func box(w, h int, border, color, title, gradient string) string {
	if w < 2 || h < 2 {
		return strings.Repeat(" ", max(w, 0))
	}
	b := borderOf(border)
	if gradient != "" && color != "" {
		return gradientBox(w, h, b, color, gradient, title)
	}
	style := fg(color)
	top := b.TopLeft + strings.Repeat(b.Top, w-2) + b.TopRight
	if title != "" && w > 4 {
		t := " " + title + " "
		if r := []rune(t); len(r) > w-4 {
			t = string(r[:w-4])
		}
		fill := w - 2 - 1 - len([]rune(t))
		top = style.Render(b.TopLeft+b.Top) + style.Bold(true).Render(t) + style.Render(strings.Repeat(b.Top, max(fill, 0))+b.TopRight)
	} else {
		top = style.Render(top)
	}
	lines := []string{top}
	for i := 1; i < h-1; i++ {
		lines = append(lines, style.Render(b.Left)+strings.Repeat(" ", w-2)+style.Render(b.Right))
	}
	lines = append(lines, style.Render(b.BottomLeft+strings.Repeat(b.Bottom, w-2)+b.BottomRight))
	return strings.Join(lines, "\n")
}

// boxOf is a box with its title in the top edge and, optionally, a right label
// before the last border cell: ╭─ Title ───── right ─╮. A title that would
// leave the label no room is cut with an ellipsis.
func boxOf(w, h int, p props) string {
	if w < 2 || h < 2 || p.str("gradient") != "" && p.str("color") != "" {
		return box(w, h, p.str("border"), p.str("color"), p.str("title"), p.str("gradient"))
	}
	b := borderOf(p.str("border"))
	style := fg(p.str("color"))
	titleStyle := fg(orDefault(p.str("titleColor"), p.str("color"))).Bold(p.flag("titleBold"))
	rightStyle := fg(orDefault(p.str("rightColor"), "240"))
	title, right := p.str("title"), p.str("right")

	// top is the top edge cell by cell, with a style class for each: 0 the
	// border, 1 the title, 2 the right label.
	top := make([]rune, w)
	class := make([]int, w)
	for i := range top {
		top[i] = []rune(b.Top)[0]
	}
	top[0], top[w-1] = []rune(b.TopLeft)[0], []rune(b.TopRight)[0]
	put := func(at int, text string, c int) {
		for i, r := range []rune(text) {
			if at+i > 0 && at+i < w-1 {
				top[at+i], class[at+i] = r, c
			}
		}
	}
	if w > 4 {
		maxBlock, rightBlock := w-4, ""
		if right != "" {
			if room := w - len([]rune(right)) - 7; room >= 4 {
				maxBlock, rightBlock = room, " "+right+" "
			}
		}
		if title != "" && maxBlock >= 4 {
			block := []rune(" " + title + " ")
			if len(block) > maxBlock {
				block = append([]rune(" "), []rune(title)[:maxBlock-3]...)
				block = append(block, '…', ' ')
			}
			put(2, string(block), 1)
		}
		if rightBlock != "" {
			put(w-2-len([]rune(rightBlock)), rightBlock, 2)
		}
	}
	var line strings.Builder
	for i := 0; i < w; {
		j := i
		for j < w && class[j] == class[i] {
			j++
		}
		st := style
		switch class[i] {
		case 1:
			st = titleStyle
		case 2:
			st = rightStyle
		}
		line.WriteString(st.Render(string(top[i:j])))
		i = j
	}
	lines := []string{line.String()}
	for i := 1; i < h-1; i++ {
		lines = append(lines, style.Render(b.Left)+strings.Repeat(" ", w-2)+style.Render(b.Right))
	}
	lines = append(lines, style.Render(b.BottomLeft+strings.Repeat(b.Bottom, w-2)+b.BottomRight))
	return strings.Join(lines, "\n")
}

func label(w int, p props) string {
	s := lipgloss.NewStyle().Bold(p.flag("bold")).Width(w).MaxWidth(w)
	if c := p.str("color"); c != "" {
		s = s.Foreground(lipgloss.Color(c))
	}
	if c := p.str("background"); c != "" {
		s = s.Background(lipgloss.Color(c))
	}
	switch p.str("align") {
	case "center":
		s = s.Align(lipgloss.Center)
	case "right":
		s = s.Align(lipgloss.Right)
	}
	return s.Render(p.str("text"))
}

func textInputPart(p props, w int) *part {
	m := textinput.New()
	m.Prompt = p.str("prompt")
	m.Placeholder = p.str("placeholder")
	m.SetValue(p.str("value"))
	m.SetWidth(max(w-lipgloss.Width(m.Prompt)-1, 1))
	st := m.Styles()
	st.Focused.Prompt = fg(p.str("color"))
	st.Blurred.Prompt = fg(p.str("color"))
	m.SetStyles(st)
	return &part{
		keys:   true,
		update: func(msg tea.Msg) tea.Cmd { var c tea.Cmd; m, c = m.Update(msg); return c },
		view:   func() string { return m.View() },
		focus: func(on bool) tea.Cmd {
			if on {
				return m.Focus()
			}
			m.Blur()
			return nil
		},
	}
}

// word is a list item.
type word string

func (w word) FilterValue() string { return string(w) }
func (w word) Title() string       { return string(w) }
func (w word) Description() string { return "" }

func listPart(p props, w, h int) *part {
	var items []list.Item
	for _, s := range split(p.str("items"), ",") {
		items = append(items, word(s))
	}
	delegate := list.NewDefaultDelegate()
	delegate.ShowDescription = false
	delegate.SetSpacing(0)
	m := list.New(items, delegate, w, h)
	m.Title = p.str("title")
	m.SetShowFilter(p.flag("show_filter"))
	m.SetFilteringEnabled(p.flag("show_filter"))
	m.SetShowStatusBar(p.flag("show_status"))
	m.Select(p.integer("selected", 0))
	return &part{
		keys:   true,
		update: func(msg tea.Msg) tea.Cmd { var c tea.Cmd; m, c = m.Update(msg); return c },
		view:   func() string { return m.View() },
	}
}

func tablePart(p props, w, h int) *part {
	var cols []table.Column
	names := split(p.str("columns"), ",")
	each := max((w-2*len(names))/max(len(names), 1), 3)
	for _, n := range names {
		cols = append(cols, table.Column{Title: n, Width: each})
	}
	var rows []table.Row
	for _, line := range splitRaw(p.str("rows"), ";") {
		rows = append(rows, table.Row(split(line, ",")))
	}
	st := table.DefaultStyles()
	st.Selected = st.Selected.Foreground(lipgloss.Color("0")).Background(lipgloss.Color(orDefault(p.str("color"), "212")))
	m := table.New(table.WithColumns(cols), table.WithRows(rows), table.WithHeight(h), table.WithWidth(w), table.WithStyles(st))
	m.SetCursor(p.integer("selected", 0))
	return &part{
		keys:   true,
		update: func(msg tea.Msg) tea.Cmd { var c tea.Cmd; m, c = m.Update(msg); return c },
		view:   func() string { return m.View() },
		focus: func(on bool) tea.Cmd {
			if on {
				m.Focus()
			} else {
				m.Blur()
			}
			return nil
		},
	}
}

func viewportPart(p props, w, h int) *part {
	bar := p.flag("show_scrollbar") && w > 3
	inner := w
	if bar {
		inner = w - 2
	}
	m := viewport.New(viewport.WithWidth(inner), viewport.WithHeight(h))
	m.SetContent(strings.ReplaceAll(p.str("content"), "|", string(rune(10))))
	if total := m.TotalLineCount(); total > h {
		m.SetYOffset(p.integer("percent", 0) * (total - h) / 100)
	}
	view := func() string {
		if !bar {
			return m.View()
		}
		// A scrollbar two cells in from the text: a track with a thumb whose
		// size and place follow the visible part of the content.
		lines := strings.Split(m.View(), string(rune(10)))
		total := max(m.TotalLineCount(), h)
		thumb := max(h*h/total, 1)
		top := 0
		if total > h {
			top = m.YOffset() * (h - thumb) / (total - h)
		}
		for i := range lines {
			track := "│"
			if i >= top && i < top+thumb {
				track = "█"
			}
			lines[i] = lipgloss.NewStyle().Width(inner).Render(lines[i]) + " " + fg(p.str("color")).Render(track)
		}
		return strings.Join(lines, string(rune(10)))
	}
	return &part{
		keys:   true,
		update: func(msg tea.Msg) tea.Cmd { var c tea.Cmd; m, c = m.Update(msg); return c },
		view:   view,
	}
}

func paginatorPart(p props) *part {
	m := paginator.New(paginator.WithTotalPages(max(p.integer("total", 5), 1)))
	if p.str("style") == "arabic" {
		m.Type = paginator.Arabic
	} else {
		m.Type = paginator.Dots
	}
	m.Page = max(p.integer("page", 1)-1, 0)
	return &part{
		keys:   true,
		update: func(msg tea.Msg) tea.Cmd { var c tea.Cmd; m, c = m.Update(msg); return c },
		view:   func() string { return fg(p.str("color")).Render(m.View()) },
	}
}

func progressPart(p props, w int) *part {
	opts := []progress.Option{progress.WithWidth(max(w, 4)), progress.WithColors(lipgloss.Color(orDefault(p.str("color"), "212")))}
	if !p.flag("show_percentage") {
		opts = append(opts, progress.WithoutPercentage())
	}
	m := progress.New(opts...)
	percent := float64(p.integer("percent", 0)) / 100
	// The bar is drawn at the percent as it is, not animated towards it, so
	// the first frame is the design.
	return &part{
		keys: true,
		update: func(msg tea.Msg) tea.Cmd {
			if k, ok := msg.(tea.KeyPressMsg); ok {
				switch k.String() {
				case "right", "up", "+":
					percent = min(percent+0.1, 1)
				case "left", "down", "-":
					percent = max(percent-0.1, 0)
				}
			}
			return nil
		},
		view: func() string { return m.ViewAs(percent) },
	}
}

func isToggle(k tea.KeyPressMsg) bool {
	s := k.String()
	return s == "space" || s == "enter" || s == " "
}

// clock writes d as mm:ss.t, or as mm:ss without the tenths.
func clock(d time.Duration, tenths bool) string {
	d = max(d, 0)
	text := fmt.Sprintf("%02d:%02d", int(d.Minutes()), int(d.Seconds())%60)
	if tenths {
		text += fmt.Sprintf(".%d", d.Milliseconds()/100%10)
	}
	return text
}

func stopwatchPart(p props) *part {
	m := stopwatch.New(stopwatch.WithInterval(100 * time.Millisecond))
	start := parseReading(p.str("value")) // the reading in the design; it counts on from there
	l := &part{keys: true, view: func() string { return fg(p.str("color")).Render(clock(start+m.Elapsed(), true)) }}
	if p.flag("running") {
		l.start = m.Start()
	}
	l.update = func(msg tea.Msg) tea.Cmd {
		if k, ok := msg.(tea.KeyPressMsg); ok {
			if isToggle(k) {
				return m.Toggle()
			}
			return nil
		}
		var c tea.Cmd
		m, c = m.Update(msg)
		return c
	}
	return l
}

func timerPart(p props) *part {
	m := timer.New(parseClock(p.str("value")), timer.WithInterval(100*time.Millisecond))
	l := &part{keys: true, view: func() string {
		return fg(p.str("color")).Render(strings.TrimSpace(p.str("label") + " " + clock(m.Timeout, false)))
	}}
	if p.flag("running") {
		l.start = m.Start()
	}
	l.update = func(msg tea.Msg) tea.Cmd {
		if k, ok := msg.(tea.KeyPressMsg); ok {
			if isToggle(k) {
				return m.Toggle()
			}
			return nil
		}
		var c tea.Cmd
		m, c = m.Update(msg)
		return c
	}
	return l
}

// parseClock reads "04:32" (minutes:seconds) or "1:02:03"; anything else is a minute.
func parseClock(s string) time.Duration {
	total := 0
	for _, piece := range strings.Split(strings.TrimSpace(s), ":") {
		n, err := strconv.Atoi(piece)
		if err != nil {
			return time.Minute
		}
		total = total*60 + n
	}
	if total == 0 {
		return time.Minute
	}
	return time.Duration(total) * time.Second
}

// gradientBox is box with a border that fades from one colour to another,
// left to right, using Lip Gloss colour blending.
func gradientBox(w, h int, b lipgloss.Border, from, to, title string) string {
	stops := lipgloss.Blend1D(w, lipgloss.Color(from), lipgloss.Color(to))
	paint := func(x int) lipgloss.Style { return lipgloss.NewStyle().Foreground(stops[min(x, len(stops)-1)]) }
	titleRunes := []rune(" " + title + " ")
	if title == "" || w <= 4 {
		titleRunes = nil
	} else if len(titleRunes) > w-4 {
		titleRunes = titleRunes[:w-4]
	}
	edge := func(left, mid, right string, withTitle bool) string {
		var s strings.Builder
		for x := 0; x < w; x++ {
			switch {
			case x == 0:
				s.WriteString(paint(x).Render(left))
			case x == w-1:
				s.WriteString(paint(x).Render(right))
			case withTitle && x >= 2 && x-2 < len(titleRunes):
				s.WriteString(paint(x).Bold(true).Render(string(titleRunes[x-2])))
			default:
				s.WriteString(paint(x).Render(mid))
			}
		}
		return s.String()
	}
	lines := []string{edge(b.TopLeft, b.Top, b.TopRight, true)}
	for i := 1; i < h-1; i++ {
		lines = append(lines, paint(0).Render(b.Left)+strings.Repeat(" ", w-2)+paint(w-1).Render(b.Right))
	}
	return strings.Join(append(lines, edge(b.BottomLeft, b.Bottom, b.BottomRight, false)), "\n")
}

// listBlock is a Lip Gloss list with the chosen enumerator.
func listBlock(w, h int, p props) string {
	var items []any
	for _, s := range split(p.str("items"), ",") {
		items = append(items, s)
	}
	enumerators := map[string]llist.Enumerator{
		"bullet": llist.Bullet, "arabic": llist.Arabic, "alphabet": llist.Alphabet, "dash": llist.Dash,
		"none": func(llist.Items, int) string { return "" },
	}
	enumerator, ok := enumerators[p.str("enumerator")]
	if !ok {
		enumerator = llist.Bullet
	}
	l := llist.New(items...).Enumerator(enumerator).
		EnumeratorStyle(fg(p.str("color")).PaddingRight(1)).ItemStyle(fg(p.str("color")))
	return lipgloss.NewStyle().MaxWidth(w).MaxHeight(h).Render(l.String())
}

// treePart is the Bubbles tree: items are indented two spaces per level and a
// line with deeper lines below it is a branch.
func treePart(p props, w, h int) *part {
	type line struct {
		name  string
		depth int
	}
	var lines []line
	for _, raw := range strings.Split(p.str("items"), "|") {
		name := strings.TrimLeft(raw, " ")
		if strings.TrimSpace(name) != "" {
			lines = append(lines, line{strings.TrimSpace(name), (len(raw) - len(name)) / 2})
		}
	}
	// nodes builds the nodes of the lines from index from, at the given depth or deeper.
	var nodes func(from, depth int) ([]any, int)
	nodes = func(from, depth int) ([]any, int) {
		var out []any
		i := from
		for i < len(lines) && lines[i].depth >= depth {
			l := lines[i]
			kids, next := nodes(i+1, l.depth+1)
			if len(kids) > 0 {
				out = append(out, tree.Root(l.name).Child(kids...))
			} else {
				out = append(out, l.name)
			}
			i = next
		}
		return out, i
	}
	kids, _ := nodes(0, 0)
	m := tree.New(tree.Root(orDefault(p.str("root"), "tree")).Child(kids...), w, h)
	m.SetShowHelp(p.flag("show_help"))
	m.SetSize(w, h)
	st := tree.DefaultDarkStyles()
	if c := p.str("color"); c != "" {
		st.SelectedNodeStyle = st.SelectedNodeStyle.Foreground(lipgloss.Color(c))
		st.CursorStyle = st.CursorStyle.Foreground(lipgloss.Color(c))
	}
	m.SetStyles(st)
	for i := 0; i < p.integer("selected", 0); i++ {
		m.Down()
	}
	return &part{
		keys:   true,
		update: func(msg tea.Msg) tea.Cmd { var c tea.Cmd; m, c = m.Update(msg); return c },
		view:   func() string { return m.View() },
	}
}

// frame is a border with the title set into the top edge at the left, centre
// or right, and the content wrapped inside.
func frame(w, h int, p props) string {
	if w < 2 || h < 2 {
		return strings.Repeat(" ", max(w, 0))
	}
	b := borderOf(p.str("border"))
	style := fg(p.str("color"))
	top := style.Render(b.TopLeft + strings.Repeat(b.Top, w-2) + b.TopRight)
	if t := p.str("title"); t != "" {
		title := " " + t + " "
		if room := w - 2; len([]rune(title)) <= room {
			slack := room - len([]rune(title))
			left := 0
			switch p.str("position") {
			case "center":
				left = slack / 2
			case "right":
				left = slack
			}
			top = style.Render(b.TopLeft+strings.Repeat(b.Top, left)) + style.Bold(true).Render(title) +
				style.Render(strings.Repeat(b.Top, slack-left)+b.TopRight)
		}
	}
	inner := lipgloss.NewStyle().Width(w - 2).Height(h - 2).MaxHeight(h - 2).Render(p.str("content"))
	lines := []string{top}
	for _, l := range strings.Split(inner, "\n") {
		lines = append(lines, style.Render(b.Left)+l+style.Render(b.Right))
	}
	return strings.Join(append(lines, style.Render(b.BottomLeft+strings.Repeat(b.Bottom, w-2)+b.BottomRight)), "\n")
}

// dialog is a rounded box with centred text and a row of buttons, the active
// one inverted.
func dialog(w, h int, p props) string {
	buttons := split(p.str("buttons"), ",")
	active := min(max(p.integer("active", 0), 0), max(len(buttons)-1, 0))
	var row []string
	for i, label := range buttons {
		s := lipgloss.NewStyle().Padding(0, 2).Foreground(lipgloss.Color("255")).Background(lipgloss.Color("238"))
		if i == active {
			s = s.Foreground(lipgloss.Color("0")).Background(lipgloss.Color("255"))
		}
		row = append(row, s.Render(label))
	}
	body := lipgloss.JoinVertical(lipgloss.Center, p.str("text"), "", lipgloss.JoinHorizontal(lipgloss.Top, row...))
	return lipgloss.NewStyle().
		Border(lipgloss.RoundedBorder()).BorderForeground(lipgloss.Color(orDefault(p.str("color"), "212"))).
		Width(max(w, 1)).Height(max(h, 1)).Align(lipgloss.Center).AlignVertical(lipgloss.Center).
		MaxWidth(w).MaxHeight(h).Render(body)
}

// statusMessagePart is a one-line message styled by level, with an optional
// spinner after the text.
func statusMessagePart(p props, w int) *part {
	style := lipgloss.NewStyle().Foreground(lipgloss.Color("252")).Background(lipgloss.Color("236"))
	switch p.str("level") {
	case "warning":
		style = lipgloss.NewStyle().Foreground(lipgloss.Color("0")).Background(lipgloss.Color("220"))
	case "error":
		style = lipgloss.NewStyle().Foreground(lipgloss.Color("255")).Background(lipgloss.Color("160"))
	}
	style = style.Width(w).MaxWidth(w).MaxHeight(1)
	m := spinner.New(spinner.WithSpinner(spinner.Dot))
	out := &part{view: func() string {
		text := " " + p.str("text")
		if p.flag("spinner") {
			text += " " + m.View()
		}
		return style.Render(text)
	}}
	out.update = func(msg tea.Msg) tea.Cmd { return nil }
	if p.flag("spinner") {
		out.start = m.Tick
		out.update = func(msg tea.Msg) tea.Cmd { var c tea.Cmd; m, c = m.Update(msg); return c }
	}
	return out
}

// toast is a BubbleUp-style notification: a rounded border and a symbol and
// message in the colour of the kind.
func toast(w, h int, p props) string {
	kinds := map[string][3]string{
		"info":  {"#00ff00", "ⓘ", "(i)"},
		"warn":  {"#ffff00", "⚠", "(!)"},
		"error": {"#ff0000", "✘", "[!!]"},
		"debug": {"#ff00ff", "?", "(?)"},
	}
	k, ok := kinds[p.str("kind")]
	if !ok {
		k = kinds["info"]
	}
	prefix := k[1]
	if p.str("symbols") == "ascii" {
		prefix = k[2]
	}
	return lipgloss.NewStyle().
		Border(lipgloss.RoundedBorder()).BorderForeground(lipgloss.Color(k[0])).Foreground(lipgloss.Color(k[0])).
		Padding(0, 1).Width(max(w, 1)).MaxWidth(w).MaxHeight(h).Render(prefix + " " + p.str("message"))
}

// parseReading reads "12.3", "00:12.3" or "1:02:03.5" as a duration; anything
// else is zero.
func parseReading(s string) time.Duration {
	seconds := 0.0
	for _, piece := range strings.Split(strings.TrimSpace(s), ":") {
		n, err := strconv.ParseFloat(piece, 64)
		if err != nil {
			return 0
		}
		seconds = seconds*60 + n
	}
	return time.Duration(seconds * float64(time.Second))
}

// swatchBlock is a colour shown as a block with its value and a label: the
// first line holds all three, further lines only the block.
func swatchBlock(h int, p props) string {
	text := fg(p.str("textColor"))
	label := p.str("label")
	x := 0
	var first strings.Builder
	if label != "" {
		w := p.integer("labelWidth", 0)
		if w <= 0 {
			w = lipgloss.Width(label) + 2
		}
		first.WriteString(text.Render(label))
		first.WriteString(strings.Repeat(" ", max(w-lipgloss.Width(label), 0)))
		x = max(w, lipgloss.Width(label))
	}
	width := max(p.integer("swatch", 2), 1)
	block := lipgloss.NewStyle().Background(lipgloss.Color(p.str("color"))).Render(strings.Repeat(" ", width))
	first.WriteString(block)
	if p.flag("showValue") {
		first.WriteString(" " + text.Render(p.str("color")))
	}
	lines := []string{first.String()}
	for i := 1; i < max(h, 1); i++ {
		lines = append(lines, strings.Repeat(" ", x)+block)
	}
	return strings.Join(lines, "\n")
}
