package main

import (
	"fmt"
	"os"

	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
)

// model is the whole program: the components of layout.go, running, and which
// of them has the keyboard.
type model struct {
	parts []*part
	// focus is the index of the component that reads the keyboard, or -1.
	focus int
	// w and h are the size of the window, once the terminal has said.
	w, h int
	// grab is the component being dragged or resized with the mouse, or nil.
	grab *grab
}

// grab is a drag in progress: which component, whether it is being resized,
// and where inside it the mouse took hold.
type grab struct {
	i      int
	resize bool
	dx, dy int
}

func newModel() model {
	m := model{focus: -1, w: canvasW, h: canvasH}
	for _, c := range layout {
		m.parts = append(m.parts, build(c))
	}
	return m
}

func (m model) Init() tea.Cmd {
	var starts []tea.Cmd
	for _, p := range m.parts {
		starts = append(starts, p.start)
	}
	return tea.Batch(starts...)
}

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		cmds := []tea.Cmd{m.resize(msg.Width, msg.Height)}
		for _, p := range m.parts {
			cmds = append(cmds, p.update(msg))
		}
		return m, tea.Batch(cmds...)
	case tea.KeyPressMsg:
		switch msg.String() {
		case "ctrl+c", "esc":
			return m, tea.Quit
		case "tab":
			return m, m.cycle(1)
		case "shift+tab":
			return m, m.cycle(-1)
		}
		if m.focus >= 0 {
			return m, m.parts[m.focus].update(msg)
		}
		return m, nil
	case tea.MouseClickMsg:
		// A press on a tab or a section header shows that page.
		if cmd, ok := m.pressHeader(msg.X, msg.Y); ok {
			return m, cmd
		}
		m.take(msg.X, msg.Y)
		return m, m.focusAt(msg.X, msg.Y)
	case tea.MouseMotionMsg:
		if m.grab != nil {
			return m, m.drag(msg.X, msg.Y)
		}
		return m, nil
	case tea.MouseReleaseMsg:
		m.grab = nil
		return m, nil
	case tea.MouseWheelMsg:
		if i := m.at(msg.X, msg.Y); i >= 0 {
			code := tea.KeyDown
			if msg.Button == tea.MouseWheelUp {
				code = tea.KeyUp
			}
			return m, m.parts[i].update(tea.KeyPressMsg{Code: code})
		}
		return m, nil
	}
	// Everything else (timer ticks, cursor blinks) goes to every component; each
	// one ignores what is not its own.
	var cmds []tea.Cmd
	for _, p := range m.parts {
		cmds = append(cmds, p.update(msg))
	}
	return m, tea.Batch(cmds...)
}

// resize places again every component that follows the window, for a window
// of w by h cells. A component whose rectangle changed is built again, so it
// starts from its initial state.
func (m *model) resize(w, h int) tea.Cmd {
	m.w, m.h = w, h
	var cmds []tea.Cmd
	for i := range layout {
		c := &layout[i]
		if c.Fit == nil {
			continue
		}
		x, y, cw, ch := c.Fit(w, h)
		cw, ch = cw+c.Dw, ch+c.Dh
		x, y = x+c.Dx, y+c.Dy
		cw, ch = min(max(cw, c.MinW), max(w, 1)), min(max(ch, c.MinH), max(h, 1))
		x, y = min(max(x, 0), max(w-cw, 0)), min(max(y, 0), max(h-ch, 0))
		if x == c.X && y == c.Y && cw == c.W && ch == c.H {
			continue
		}
		oldX, oldY, oldW, oldH := c.X, c.Y, c.W, c.H
		c.X, c.Y, c.W, c.H = x, y, cw, ch
		m.moveMembers(i, oldX, oldY, oldW, oldH)
		cmds = append(cmds, m.rebuild(i))
	}
	return tea.Batch(cmds...)
}

// rebuild makes component i again for its rectangle and gives it the keyboard
// if it had it.
func (m *model) rebuild(i int) tea.Cmd {
	m.parts[i] = build(layout[i])
	cmds := []tea.Cmd{m.parts[i].start}
	if i == m.focus && m.parts[i].focus != nil {
		cmds = append(cmds, m.parts[i].focus(true))
	}
	return tea.Batch(cmds...)
}

// bounds is the area components can be moved in: the window when the design
// follows it, otherwise the canvas.
func (m model) bounds() (int, int) {
	if responsive {
		return m.w, m.h
	}
	return canvasW, canvasH
}

// take starts a drag when the click lands on a component the person may move
// (anywhere on it) or resize (its bottom-right cell).
func (m *model) take(x, y int) {
	i := m.at(x, y)
	if i < 0 {
		return
	}
	c := layout[i]
	switch {
	case c.Resize && x == c.X+c.W-1 && y == c.Y+c.H-1:
		m.grab = &grab{i: i, resize: true}
	case c.Drag:
		m.grab = &grab{i: i, dx: x - c.X, dy: y - c.Y}
	}
}

// drag follows the mouse while a component is being moved or resized.
func (m *model) drag(x, y int) tea.Cmd {
	g := m.grab
	c := &layout[g.i]
	w, h := m.bounds()
	oldX, oldY, oldW, oldH := c.X, c.Y, c.W, c.H
	if g.resize {
		c.W = min(max(x-c.X+1, c.MinW, 1), max(w-c.X, 1))
		c.H = min(max(y-c.Y+1, c.MinH, 1), max(h-c.Y, 1))
	} else {
		c.X = min(max(x-g.dx, 0), max(w-c.W, 0))
		c.Y = min(max(y-g.dy, 0), max(h-c.H, 0))
	}
	if c.X == oldX && c.Y == oldY && c.W == oldW && c.H == oldH {
		return nil
	}
	c.Dx += c.X - oldX
	c.Dy += c.Y - oldY
	c.Dw += c.W - oldW
	c.Dh += c.H - oldH
	m.moveMembers(g.i, oldX, oldY, oldW, oldH)
	return m.rebuild(g.i)
}

// cycle moves the keyboard to the next (or previous) component that reads it.
func (m *model) cycle(step int) tea.Cmd {
	n := len(m.parts)
	for i := 1; i <= n; i++ {
		next := ((m.focus+step*i)%n + n) % n
		if m.parts[next].keys && m.shown(next) {
			return m.setFocus(next)
		}
	}
	return nil
}

func (m *model) setFocus(i int) tea.Cmd {
	if i == m.focus {
		return nil
	}
	var cmds []tea.Cmd
	if m.focus >= 0 && m.parts[m.focus].focus != nil {
		cmds = append(cmds, m.parts[m.focus].focus(false))
	}
	m.focus = i
	if i >= 0 && m.parts[i].focus != nil {
		cmds = append(cmds, m.parts[i].focus(true))
	}
	return tea.Batch(cmds...)
}

// at is the top-most component under the cell, or -1.
func (m model) at(x, y int) int {
	for i := len(layout) - 1; i >= 0; i-- {
		c := layout[i]
		if m.shown(i) && x >= c.X && x < c.X+c.W && y >= c.Y && y < c.Y+c.H {
			return i
		}
	}
	return -1
}

// shown reports whether component i is drawn: a member only while its
// container is shown and its page is the shown tab or an open section.
func (m model) shown(i int) bool {
	c := layout[i]
	if !c.Member {
		return true
	}
	cont := layout[c.Of]
	return m.shown(c.Of) && containerShown(cont.Kind, props(cont.Props), c.Page)
}

// pressHeader handles a click on a tab or a section header of a container: it
// changes the page that is shown and moves the members with their bodies.
func (m *model) pressHeader(x, y int) (tea.Cmd, bool) {
	i := m.at(x, y)
	if i < 0 || (layout[i].Kind != "cuppa.tabs" && layout[i].Kind != "cuppa.panels") {
		return nil, false
	}
	c := &layout[i]
	page, ok := containerHeaderAt(c.Kind, props(c.Props), c.W, c.H, x-c.X, y-c.Y)
	if !ok {
		return nil, false
	}
	oldX, oldY, oldW, oldH := c.X, c.Y, c.W, c.H
	before := containerGeometry(c.Kind, props(c.Props), c.W, c.H)
	for k, v := range containerPressed(c.Kind, props(c.Props), page) {
		c.Props[k] = v
	}
	m.shiftMembers(i, before, oldX, oldY, oldW, oldH)
	return m.rebuild(i), true
}

// moveMembers moves the members of container i after the container itself
// moved or was resized from the old rectangle.
func (m *model) moveMembers(i, oldX, oldY, oldW, oldH int) {
	c := layout[i]
	if c.Kind != "cuppa.tabs" && c.Kind != "cuppa.panels" {
		return
	}
	before := containerGeometry(c.Kind, props(c.Props), oldW, oldH)
	m.shiftMembers(i, before, oldX, oldY, oldW, oldH)
}

// shiftMembers moves each member by how far the top-left of its page's body
// moved between before (at the old position) and now.
func (m *model) shiftMembers(i int, before containerGeo, oldX, oldY, oldW, oldH int) {
	c := layout[i]
	after := containerGeometry(c.Kind, props(c.Props), c.W, c.H)
	for j := range layout {
		member := &layout[j]
		if !member.Member || member.Of != i || member.Page >= len(after.bodies) || member.Page >= len(before.bodies) {
			continue
		}
		member.X += (c.X + after.bodies[member.Page][0]) - (oldX + before.bodies[member.Page][0])
		member.Y += (c.Y + after.bodies[member.Page][1]) - (oldY + before.bodies[member.Page][1])
	}
}

func (m *model) focusAt(x, y int) tea.Cmd {
	i := m.at(x, y)
	if i < 0 || !m.parts[i].keys {
		i = -1
	}
	return m.setFocus(i)
}

func (m model) View() tea.View {
	w, h := canvasW, canvasH
	if responsive {
		w, h = m.w, m.h
	}
	layers := []*lipgloss.Layer{lipgloss.NewLayer(backdrop(w, h)).Z(0)}
	for i, c := range layout {
		if !m.shown(i) {
			continue
		}
		layers = append(layers, lipgloss.NewLayer(m.parts[i].view()).X(c.X).Y(c.Y).Z(i+1))
	}
	v := tea.NewView(lipgloss.NewCompositor(layers...).Render())
	v.AltScreen = true
	v.MouseMode = tea.MouseModeCellMotion
	return v
}

// backdrop is the canvas itself: its size and background colour.
func backdrop(w, h int) string {
	row := make([]byte, max(w, 1))
	for i := range row {
		row[i] = ' '
	}
	line := string(row)
	if background != "" {
		line = lipgloss.NewStyle().Background(lipgloss.Color(background)).Render(line)
	}
	lines := make([]string, max(h, 1))
	for i := range lines {
		lines[i] = line
	}
	return joinLines(lines)
}

func joinLines(lines []string) string {
	out := ""
	for i, l := range lines {
		if i > 0 {
			out += "\n"
		}
		out += l
	}
	return out
}

func main() {
	if _, err := tea.NewProgram(newModel()).Run(); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}
