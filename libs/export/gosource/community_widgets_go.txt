// approximated here: Dropdown, Prompt input, Prompt select, Data tree, PDF

func widgetPart(draw func(g *sketchGrid, p props)) func(c placed, p props) *part {
	return func(c placed, p props) *part {
		g := newSketchGrid(c.W, c.H)
		draw(g, p)
		return staticPart(g.String())
	}
}

var sketchDim = sketchStyle{Fg: "240"}

func sketchSelected(color string) sketchStyle { return sketchStyle{Fg: "0", Bg: color, Bold: true} }

// sketchLines splits a value on "|", the line separator of single-line property
// editors, keeping the indentation.
func sketchLines(s string) []string {
	if s == "" {
		return nil
	}
	parts := strings.Split(s, "|")
	for i := range parts {
		parts[i] = strings.TrimRight(parts[i], " ")
	}
	return parts
}

type sketchStyle struct {
	Fg, Bg  string
	Bold    bool
	Dim     bool
	Reverse bool
}

type sketchCell struct {
	Ch    rune
	Style sketchStyle
}

type sketchRect struct{ X, Y, W, H int }

type sketchBorder struct{ TL, TR, BL, BR, H, V rune }

func sketchBorderNamed(string) sketchBorder { return sketchBorder{'╭', '╮', '╰', '╯', '─', '│'} }

// sketchGrid is a character grid the drawings above are written against.
type sketchGrid struct {
	W, H int
	c    []sketchCell
}

func newSketchGrid(w, h int) *sketchGrid {
	w, h = max(w, 0), max(h, 0)
	return &sketchGrid{W: w, H: h, c: make([]sketchCell, w*h)}
}

func (g *sketchGrid) Set(x, y int, c sketchCell) {
	if x >= 0 && y >= 0 && x < g.W && y < g.H {
		g.c[y*g.W+x] = c
	}
}

func (g *sketchGrid) Text(x, y int, text string, s sketchStyle, maxW int) int {
	n := 0
	for _, ch := range text {
		w := sketchRuneWidth(ch)
		if w == 0 {
			continue
		}
		if maxW > 0 && n+w > maxW || w == 2 && x+n+w > g.W {
			break
		}
		g.Set(x+n, y, sketchCell{Ch: ch, Style: s})
		if w == 2 {
			g.Set(x+n+1, y, sketchCell{Ch: sketchWideTail, Style: s})
		}
		n += w
	}
	return n
}

// sketchWideTail is the second cell of a double-width character.
const sketchWideTail rune = -1

// sketchRuneWidth is the number of terminal cells of a rune: 0, 1 or 2.
func sketchRuneWidth(r rune) int {
	if r < 0x300 {
		return 1
	}
	return lipgloss.Width(string(r))
}

// sketchWidth is the number of terminal cells of a text.
func sketchWidth(s string) int { return lipgloss.Width(s) }

func (g *sketchGrid) Fill(r sketchRect, ch rune, s sketchStyle) {
	for y := r.Y; y < r.Y+r.H; y++ {
		for x := r.X; x < r.X+r.W; x++ {
			g.Set(x, y, sketchCell{Ch: ch, Style: s})
		}
	}
}

func (g *sketchGrid) Box(r sketchRect, b sketchBorder, s sketchStyle) {
	g.Fill(r, ' ', s)
	if r.W < 2 || r.H < 2 {
		return
	}
	right, bottom := r.X+r.W-1, r.Y+r.H-1
	for x := r.X + 1; x < right; x++ {
		g.Set(x, r.Y, sketchCell{Ch: b.H, Style: s})
		g.Set(x, bottom, sketchCell{Ch: b.H, Style: s})
	}
	for y := r.Y + 1; y < bottom; y++ {
		g.Set(r.X, y, sketchCell{Ch: b.V, Style: s})
		g.Set(right, y, sketchCell{Ch: b.V, Style: s})
	}
	g.Set(r.X, r.Y, sketchCell{Ch: b.TL, Style: s})
	g.Set(right, r.Y, sketchCell{Ch: b.TR, Style: s})
	g.Set(r.X, bottom, sketchCell{Ch: b.BL, Style: s})
	g.Set(right, bottom, sketchCell{Ch: b.BR, Style: s})
}

// String renders the rows, styling each run of sketchGrid that share a style once.
func (g *sketchGrid) String() string {
	var out strings.Builder
	for y := 0; y < g.H; y++ {
		if y > 0 {
			out.WriteByte('\n')
		}
		var run strings.Builder
		var cur sketchStyle
		flush := func() {
			if run.Len() == 0 {
				return
			}
			text := run.String()
			if cur != (sketchStyle{}) {
				st := lipgloss.NewStyle().Bold(cur.Bold).Faint(cur.Dim).Reverse(cur.Reverse)
				if cur.Fg != "" {
					st = st.Foreground(lipgloss.Color(cur.Fg))
				}
				if cur.Bg != "" {
					st = st.Background(lipgloss.Color(cur.Bg))
				}
				text = st.Render(text)
			}
			out.WriteString(text)
			run.Reset()
		}
		for x := 0; x < g.W; x++ {
			c := g.c[y*g.W+x]
			if c.Ch == sketchWideTail {
				continue
			}
			if c.Style != cur {
				flush()
				cur = c.Style
			}
			ch := c.Ch
			if ch == 0 {
				ch = ' '
			}
			run.WriteRune(ch)
		}
		flush()
	}
	return out.String()
}

// drawDropdown draws bubble-dropdown's "[ Label ▼ ]" trigger and, when open,
// its panel below: a rounded box with the options, the chosen one inverted.
func drawDropdown(g *sketchGrid, p props) {
	accent := p.str("color")
	options := split(p.str("options"), ",")
	sel := p.integer("selected", -1)
	label, labelStyle := p.str("placeholder"), sketchDim
	if sel >= 0 && sel < len(options) {
		label, labelStyle = options[sel], sketchStyle{}
	}
	x := g.Text(0, 0, "[ ", sketchStyle{}, g.W)
	x += g.Text(x, 0, label, labelStyle, g.W-x)
	x += g.Text(x, 0, " ", sketchStyle{}, g.W-x)
	x += g.Text(x, 0, "▼", sketchStyle{Fg: accent, Bold: true}, g.W-x)
	g.Text(x, 0, " ]", sketchStyle{}, g.W-x)
	if !p.flag("open") || g.H < 3 || len(options) == 0 {
		return
	}
	longest := 0
	for _, o := range options {
		longest = max(longest, len([]rune(o)))
	}
	triggerW := len([]rune(label)) + 6
	w := min(g.W, max(triggerW, longest+4))
	rows := min(len(options), g.H-3)
	if rows < 1 {
		return
	}
	start := 0
	if sel >= rows {
		start = min(sel-rows+1, len(options)-rows)
	}
	g.Box(sketchRect{X: 0, Y: 1, W: w, H: rows + 2}, sketchBorderNamed("rounded"), sketchStyle{Fg: accent})
	for i := 0; i < rows; i++ {
		style := sketchStyle{}
		if start+i == sel {
			style = sketchSelected(accent)
		}
		g.Fill(sketchRect{X: 1, Y: 2 + i, W: w - 2, H: 1}, ' ', style)
		g.Text(2, 2+i, options[start+i], style, w-4)
	}
}

// drawPromptInput draws promptkit's text input: the prompt, the value (or the
// placeholder) and the cursor, and under it the validation error.
func drawPromptInput(g *sketchGrid, p props) {
	accent := p.str("color")
	x := g.Text(0, 0, p.str("prompt"), sketchStyle{Fg: accent, Bold: true}, g.W)
	x += g.Text(x, 0, " ", sketchStyle{}, g.W-x)
	cursor := sketchStyle{Fg: accent}
	value := p.str("value")
	switch {
	case value == "":
		x += g.Text(x, 0, "█", cursor, g.W-x)
		g.Text(x, 0, p.str("placeholder"), sketchDim, g.W-x)
	case p.flag("hidden"):
		x += g.Text(x, 0, strings.Repeat("•", len([]rune(value))), sketchStyle{}, g.W-x)
		g.Text(x, 0, "█", cursor, g.W-x)
	default:
		x += g.Text(x, 0, value, sketchStyle{}, g.W-x)
		g.Text(x, 0, "█", cursor, g.W-x)
	}
	if e := p.str("error"); e != "" && g.H >= 2 {
		g.Text(0, 1, "Error: "+e, sketchStyle{Fg: "196"}, g.W)
	}
}

// drawPromptSelect draws promptkit's selection prompt: the prompt with the
// filter after it, then the choices that match, a cursor on the chosen one and
// arrows where more choices are out of view.
func drawPromptSelect(g *sketchGrid, p props) {
	accent := p.str("color")
	x := g.Text(0, 0, p.str("prompt"), sketchStyle{Fg: accent, Bold: true}, g.W)
	filter := p.str("filter")
	if filter != "" {
		x += g.Text(x, 0, " "+filter, sketchStyle{}, g.W-x)
		g.Text(x, 0, "█", sketchStyle{Fg: accent}, g.W-x)
	}
	var choices []string
	for _, c := range split(p.str("choices"), ",") {
		if filter == "" || strings.Contains(strings.ToLower(c), strings.ToLower(filter)) {
			choices = append(choices, c)
		}
	}
	rows := g.H - 1
	if rows < 1 || len(choices) == 0 {
		return
	}
	sel := min(max(p.integer("selected", 0), 0), len(choices)-1)
	start := 0
	if len(choices) > rows {
		start = min(max(sel-rows/2, 0), len(choices)-rows)
	}
	end := min(start+rows, len(choices))
	for i := 0; start+i < end; i++ {
		idx := start + i
		prefix, style := "  ", sketchStyle{}
		switch {
		case idx == sel:
			prefix, style = "▸ ", sketchStyle{Fg: accent, Bold: true}
		case i == 0 && start > 0:
			prefix = "⇡ "
		case start+i == end-1 && end < len(choices):
			prefix = "⇣ "
		}
		cx := g.Text(0, 1+i, prefix, style, g.W)
		g.Text(cx, 1+i, choices[idx], style, g.W-cx)
	}
}

// drawDataTree draws bubble-data-tree: a data structure as a tree of
// "key: value" lines, nested by two spaces of indentation.
func drawDataTree(g *sketchGrid, p props) {
	accent := p.str("color")
	lines := sketchLines(p.str("data"))
	depths := make([]int, len(lines))
	texts := make([]string, len(lines))
	for i, l := range lines {
		trimmed := strings.TrimLeft(l, " ")
		depths[i], texts[i] = (len(l)-len(trimmed))/2, trimmed
	}
	hasLater := func(i int) bool {
		for j := i + 1; j < len(lines); j++ {
			if depths[j] == depths[i] {
				return true
			}
			if depths[j] < depths[i] {
				return false
			}
		}
		return false
	}
	for i := range lines {
		if i >= g.H {
			break
		}
		x := 0
		for level := 1; level < depths[i]; level++ {
			ancestor := -1
			for k := i - 1; k >= 0; k-- {
				if depths[k] <= level {
					if depths[k] == level {
						ancestor = k
					}
					break
				}
			}
			mark := "   "
			if ancestor >= 0 && hasLater(ancestor) {
				mark = "│  "
			}
			x += g.Text(x, i, mark, sketchDim, g.W-x)
		}
		if depths[i] > 0 {
			branch := "├─ "
			if !hasLater(i) {
				branch = "└─ "
			}
			x += g.Text(x, i, branch, sketchDim, g.W-x)
		}
		key, value, found := strings.Cut(texts[i], ": ")
		if !found {
			g.Text(x, i, texts[i], sketchStyle{Fg: accent, Bold: true}, g.W-x)
			continue
		}
		x += g.Text(x, i, key, sketchStyle{Fg: accent}, g.W-x)
		x += g.Text(x, i, ": ", sketchDim, g.W-x)
		g.Text(x, i, value, sketchStyle{}, g.W-x)
	}
}

// drawPDFView draws ntcharts-pdf: a light page with the text of the document
// (text mode) or the same lines greeked into bars (image mode), and a status
// line with the file, the page and the mode.
func drawPDFView(g *sketchGrid, p props) {
	if g.H < 2 {
		g.Text(0, 0, p.str("path"), sketchDim, g.W)
		return
	}
	page := sketchStyle{Fg: "234", Bg: "255"}
	g.Fill(sketchRect{W: g.W, H: g.H - 1}, ' ', page)
	image := p.str("mode") == "image"
	for i, l := range sketchLines(p.str("text")) {
		if i >= g.H-1 {
			break
		}
		if image {
			l = strings.Map(func(r rune) rune {
				if r == ' ' {
					return ' '
				}
				return '▬'
			}, l)
		}
		g.Text(2, i, l, page, g.W-3)
	}
	status := " " + p.str("path") + " · page " + strconv.Itoa(p.integer("page", 1)) + "/" + strconv.Itoa(p.integer("pages", 1)) + " · " + p.str("mode") + " mode"
	g.Text(0, g.H-1, status, sketchDim, g.W)
}

// wave3D is the height pattern the 3D chart's series follow.
var wave3D = [...]int{2, 3, 4, 6, 5, 4, 3, 2}

// drawChart3D draws ntcharts3d: three axes (z up, x across, y into the
// screen), the series of the chosen kind in the plot, a title and a legend.
func drawChart3D(g *sketchGrid, p props) {
	accent := p.str("color")
	top := 0
	if t := p.str("title"); t != "" && g.H >= 6 {
		g.Text(0, 0, t, sketchStyle{Bold: true}, g.W)
		top = 1
	}
	kind := p.str("kind")
	if p.flag("legend") && g.W >= 24 {
		label := "● " + kind
		g.Text(g.W-len([]rune(label)), top, label, sketchStyle{Fg: accent}, g.W)
	}
	floor := g.H - 3
	if floor <= top+1 || g.W < 8 {
		g.Text(0, 0, "3D chart", sketchDim, g.W)
		return
	}
	axis := sketchDim
	for y := top + 1; y < floor; y++ {
		g.Set(2, y, sketchCell{Ch: '│', Style: axis})
	}
	g.Set(2, floor, sketchCell{Ch: '└', Style: axis})
	for x := 3; x < g.W-1; x++ {
		g.Set(x, floor, sketchCell{Ch: '─', Style: axis})
	}
	g.Set(1, floor+1, sketchCell{Ch: '╱', Style: axis})
	g.Set(0, floor+2, sketchCell{Ch: 'y', Style: axis})
	g.Set(3, top, sketchCell{Ch: 'z', Style: axis})
	g.Set(g.W-1, floor, sketchCell{Ch: 'x', Style: axis})

	left, right := 4, g.W-2
	height := floor - top - 1
	series := sketchStyle{Fg: accent}
	column := func(x int) int { return wave3D[(x/2)%len(wave3D)] * height / 8 }
	switch kind {
	case "scatter":
		for i := 0; i < (right-left)*height/10+3; i++ {
			x := left + (i*37)%(right-left)
			y := floor - 1 - (i*53)%height
			ch := '●'
			if i%3 == 0 {
				ch = '·'
			}
			g.Set(x, y, sketchCell{Ch: ch, Style: series})
		}
	case "bar":
		for x := left; x+2 < right; x += 4 {
			h := max(wave3D[(x/4)%len(wave3D)]*height/7, 1)
			for k := 0; k < h; k++ {
				g.Set(x, floor-1-k, sketchCell{Ch: '█', Style: series})
				g.Set(x+1, floor-1-k, sketchCell{Ch: '█', Style: series})
				g.Set(x+2, floor-1-k, sketchCell{Ch: '▒', Style: series})
			}
		}
	case "line":
		prev := -1
		for x := left; x < right; x++ {
			y := floor - 1 - column(x)
			if prev >= 0 {
				for yy := min(prev, y); yy <= max(prev, y); yy++ {
					g.Set(x, yy, sketchCell{Ch: '│', Style: series})
				}
			}
			g.Set(x, y, sketchCell{Ch: '●', Style: series})
			prev = y
		}
	case "vector":
		arrows := []rune("→↗↑↖←↙↓↘")
		for y := top + 1; y < floor; y += 2 {
			for x := left; x < right; x += 4 {
				g.Set(x, y, sketchCell{Ch: arrows[(x*3+y*5)%len(arrows)], Style: series})
			}
		}
	default: // surface
		for x := left; x < right; x++ {
			h := max(column(x), 1)
			for k := 0; k < h; k++ {
				ch := '░'
				if k == h-1 {
					ch = '▓'
				}
				g.Set(x, floor-1-k, sketchCell{Ch: ch, Style: series})
			}
		}
	}
}


func (p props) float(key string, fallback float64) float64 {
	f, err := strconv.ParseFloat(strings.TrimSpace(p[key]), 64)
	if err != nil {
		return fallback
	}
	return f
}

// sgrRun is text with the style the escape sequences before it gave it.
type sgrRun struct {
	text  string
	style sketchStyle
}

// parseSGR splits text that may carry SGR escape sequences (colours, bold,
// dim, reverse) into runs of one style, starting from base. Other escape
// sequences are dropped. Text with no escape is one run.
func parseSGR(text string, base sketchStyle) []sgrRun {
	if !strings.ContainsRune(text, 0x1b) {
		return []sgrRun{{text, base}}
	}
	var runs []sgrRun
	cur := base
	var b strings.Builder
	flush := func() {
		if b.Len() > 0 {
			runs = append(runs, sgrRun{b.String(), cur})
			b.Reset()
		}
	}
	rs := []rune(text)
	for i := 0; i < len(rs); i++ {
		if rs[i] != 0x1b {
			b.WriteRune(rs[i])
			continue
		}
		// ESC [ params final: only the final byte m changes the style.
		if i+1 < len(rs) && rs[i+1] == '[' {
			j := i + 2
			for j < len(rs) && (rs[j] < 0x40 || rs[j] > 0x7e) {
				j++
			}
			if j < len(rs) {
				if rs[j] == 'm' {
					flush()
					cur = applySGR(cur, string(rs[i+2:j]))
				}
				i = j
				continue
			}
		}
		// ESC ] ... (OSC, such as a link) up to BEL or ESC \ is dropped.
		if i+1 < len(rs) && rs[i+1] == ']' {
			j := i + 2
			for j < len(rs) && rs[j] != 0x07 && (rs[j] != 0x1b || j+1 >= len(rs) || rs[j+1] != '\\') {
				j++
			}
			if j < len(rs) && rs[j] == 0x1b {
				j++
			}
			i = j
			continue
		}
	}
	flush()
	return runs
}

// applySGR changes style by the parameters of one SGR sequence. A reset (0, 39
// or 49) means the terminal's default colours, not the style the text started
// from: text that carries its own sequences is in charge of its colours.
func applySGR(style sketchStyle, params string) sketchStyle {
	if params == "" {
		return sketchStyle{}
	}
	parts := strings.Split(params, ";")
	num := func(i int) (int, bool) {
		if i >= len(parts) {
			return 0, false
		}
		n, err := strconv.Atoi(parts[i])
		return n, err == nil
	}
	colour := func(i int) (string, int) { // the colour after 38 or 48 and how many parameters it took
		mode, _ := num(i + 1)
		switch mode {
		case 5:
			n, _ := num(i + 2)
			return strconv.Itoa(n), 3
		case 2:
			r, _ := num(i + 2)
			g, _ := num(i + 3)
			bl, _ := num(i + 4)
			return "#" + hex2(r) + hex2(g) + hex2(bl), 5
		}
		return "", 1
	}
	for i := 0; i < len(parts); i++ {
		n, ok := num(i)
		if !ok {
			continue
		}
		switch {
		case n == 0:
			style = sketchStyle{}
		case n == 1:
			style.Bold = true
		case n == 2:
			style.Dim = true
		case n == 22:
			style.Bold, style.Dim = false, false
		case n == 7:
			style.Reverse = true
		case n == 27:
			style.Reverse = false
		case n >= 30 && n <= 37:
			style.Fg = strconv.Itoa(n - 30)
		case n >= 90 && n <= 97:
			style.Fg = strconv.Itoa(n - 90 + 8)
		case n >= 40 && n <= 47:
			style.Bg = strconv.Itoa(n - 40)
		case n >= 100 && n <= 107:
			style.Bg = strconv.Itoa(n - 100 + 8)
		case n == 39:
			style.Fg = ""
		case n == 49:
			style.Bg = ""
		case n == 38 || n == 48:
			c, used := colour(i)
			if c != "" {
				if n == 38 {
					style.Fg = c
				} else {
					style.Bg = c
				}
			}
			i += used - 1
		}
	}
	return style
}

func hex2(n int) string {
	const digits = "0123456789abcdef"
	n = min(max(n, 0), 255)
	return string([]byte{digits[n>>4], digits[n&15]})
}

// runesWidth is the number of cells of the text of the runs.
func runesWidth(runs []sgrRun) int {
	n := 0
	for _, r := range runs {
		n += sketchWidth(r.text)
	}
	return n
}

// cutRuns keeps the first w cells of the runs; with ellipsis a cut ends in
// "…" in the style of the last run kept.
func cutRuns(runs []sgrRun, w int, ellipsis bool) []sgrRun {
	if w <= 0 {
		return nil
	}
	total := runesWidth(runs)
	if total <= w {
		return runs
	}
	keep := w
	if ellipsis {
		keep = w - 1
	}
	var out []sgrRun
	var last sketchStyle
	for _, r := range runs {
		if keep <= 0 {
			break
		}
		var kept []rune
		for _, c := range r.text {
			w := sketchRuneWidth(c)
			if w > keep {
				keep = 0
				break
			}
			kept = append(kept, c)
			keep -= w
		}
		out = append(out, sgrRun{string(kept), r.style})
		last = r.style
	}
	if ellipsis {
		out = append(out, sgrRun{"…", last})
	}
	return out
}

// rowsColumn is one column of the Rows component: its name, its width (0 for
// the rest of the row) and how its cells are drawn.
type rowsColumn struct {
	name  string
	width int
	// colour makes the cell a block of that colour (a background); glyph makes
	// it full blocks drawn in that colour.
	colour, glyph bool
	// ellipsis cuts a cell that is too long with "…".
	ellipsis bool
	// fg and bg are the colour of the column's text and its background; selBg,
	// selFg and selBold are what the column takes on a selected row. They are
	// colours or @Name of the palette.
	fg, bg, selBg, selFg string
	selBold              bool
	// fit limits the selected background to the cells the text uses, not the
	// column's whole width.
	fit bool
	// fill takes what the other columns leave (shared by the fill columns), and
	// auto is as wide as the text of its own cell in that row: columns after a
	// fill column are therefore flush with the right end, however wide an
	// auto cell is.
	fill, auto bool
	// hasFit sizes the column from its longest cell: at least fitMin, at most
	// fitMax cells, or fitPct percent of the width the fixed columns leave but
	// never below floor (fitMin when 0). hideEmpty gives it no width and no
	// gap when every cell is empty; minWidth is the least a fill column takes.
	hasFit         bool
	fitMin, fitMax int
	fitPct, floor  int
	hideEmpty      bool
	minWidth       int
}

// rowsColumns reads "Name[:width][:token]..." items separated by commas. A
// token is colour, glyph, ellipsis, fg=<colour>, bg=<colour>,
// selbg=<colour>, selfg=<colour>, selbold, fit, fill or auto. A column without a width takes twelve cells, except the
// last, which takes what is left of the row.
func rowsColumns(spec string) []rowsColumn {
	var out []rowsColumn
	for _, item := range rowsSplit(spec, ",", true) {
		parts := strings.Split(strings.TrimSpace(item), ":")
		col := rowsColumn{name: strings.TrimSpace(parts[0])}
		for _, extra := range parts[1:] {
			extra = strings.TrimSpace(extra)
			lower := strings.ToLower(extra)
			if n, err := strconv.Atoi(extra); err == nil {
				col.width = max(n, 0)
				continue
			}
			switch {
			case lower == "colour" || lower == "color":
				col.colour = true
			case lower == "glyph":
				col.glyph = true
			case lower == "ellipsis":
				col.ellipsis = true
			case lower == "fit":
				col.fit = true
			case strings.HasPrefix(lower, "fit="):
				col.hasFit = true
				spec := strings.TrimSpace(extra[4:])
				lo, hi, _ := strings.Cut(spec, "..")
				col.fitMin, _ = strconv.Atoi(strings.TrimSpace(lo))
				hi = strings.TrimSpace(hi)
				if pct, ok := strings.CutSuffix(hi, "%"); ok {
					col.fitPct, _ = strconv.Atoi(strings.TrimSpace(pct))
				} else if hi != "" {
					col.fitMax, _ = strconv.Atoi(hi)
				}
			case strings.HasPrefix(lower, "floor="):
				col.floor, _ = strconv.Atoi(strings.TrimSpace(extra[6:]))
			case strings.HasPrefix(lower, "min="):
				col.minWidth, _ = strconv.Atoi(strings.TrimSpace(extra[4:]))
			case lower == "hide-empty":
				col.hideEmpty = true
			case lower == "fill":
				col.fill = true
			case lower == "auto":
				col.auto = true
			case lower == "selbold":
				col.selBold = true
			case strings.HasPrefix(lower, "selfg="):
				col.selFg = strings.TrimSpace(extra[6:])
			case strings.HasPrefix(lower, "fg="):
				col.fg = strings.TrimSpace(extra[3:])
			case strings.HasPrefix(lower, "bg="):
				col.bg = strings.TrimSpace(extra[3:])
			case strings.HasPrefix(lower, "selbg="):
				col.selBg = strings.TrimSpace(extra[6:])
			}
		}
		out = append(out, col)
	}
	for i := range out {
		if out[i].width == 0 && i < len(out)-1 && !out[i].fill && !out[i].auto && !out[i].hasFit {
			out[i].width = 12
		}
	}
	return out
}

// rowsSplit cuts s at sep. Unlike the list splitting it keeps empty pieces and
// whitespace, because a cell may be empty or start with spaces. A backslash
// before a comma, a semicolon or another backslash is that character itself
// (unescape false keeps the backslash, for the first cut of the rows).
func rowsSplit(s, sep string, unescape bool) []string {
	var out []string
	var cur strings.Builder
	for i := 0; i < len(s); i++ {
		c := s[i]
		if c == '\\' && i+1 < len(s) && (s[i+1] == ',' || s[i+1] == ';' || s[i+1] == '\\') {
			if !unescape {
				cur.WriteByte(c)
			}
			cur.WriteByte(s[i+1])
			i++
			continue
		}
		if strings.HasPrefix(s[i:], sep) {
			out = append(out, cur.String())
			cur.Reset()
			i += len(sep) - 1
			continue
		}
		cur.WriteByte(c)
	}
	return append(out, cur.String())
}

// rowsCells reads the rows: cells separated by commas, rows by semicolons.
func rowsCells(s string) [][]string {
	var rows [][]string
	for _, raw := range rowsSplit(s, ";", false) {
		if raw == "" {
			continue
		}
		rows = append(rows, rowsSplit(raw, ",", true))
	}
	return rows
}

// rowsPlan decides the widths of the columns of all the rows: a fixed column
// as it is, a fit column from its longest cell (within its bounds, a share of
// what the fixed columns leave), a column that hide-empty hides when it has no
// text, an auto column as wide as the cell of each row, and a fill column what
// is left.
type rowsPlan struct {
	columns []rowsColumn
	visible []bool
	// shared is the width of the fixed and fit columns, -1 for the others.
	shared []int
	// used is the total of shared, gaps the cells between visible columns.
	used, gaps, total int
	gap               int
}

func newRowsPlan(columns []rowsColumn, rows [][]string, total, gap int) rowsPlan {
	p := rowsPlan{columns: columns, total: total, gap: gap}
	p.visible = make([]bool, len(columns))
	p.shared = make([]int, len(columns))
	longest := make([]int, len(columns))
	shown := 0
	for i, c := range columns {
		empty := true
		for _, r := range rows {
			if i < len(r) && r[i] != "" {
				empty = false
				longest[i] = max(longest[i], runesWidth(parseSGR(r[i], sketchStyle{})))
			}
		}
		p.visible[i] = !c.hideEmpty || !empty
		if p.visible[i] {
			shown++
		}
	}
	p.gaps = gap * max(shown-1, 0)
	for i, c := range columns {
		switch {
		case !p.visible[i]:
			p.shared[i] = 0
		case c.hasFit:
			p.shared[i] = -2
		case c.auto || c.fill || c.width == 0 && i == len(columns)-1:
			p.shared[i] = -1
		default:
			p.shared[i] = c.width
			p.used += c.width
		}
	}
	rest := max(total-p.used-p.gaps, 0)
	for i, c := range columns {
		if p.shared[i] != -2 {
			continue
		}
		upper := longest[i]
		switch {
		case c.fitPct > 0:
			upper = max(max(c.floor, c.fitMin), rest*c.fitPct/100)
		case c.fitMax > 0:
			upper = c.fitMax
		}
		w := min(max(longest[i], c.fitMin), max(upper, c.fitMin))
		p.shared[i] = w
		p.used += w
	}
	return p
}

// row is the width of each column in one row.
func (p rowsPlan) row(row []string) []int {
	widths := make([]int, len(p.columns))
	used, fills := p.used, 0
	for i, c := range p.columns {
		switch {
		case !p.visible[i]:
		case p.shared[i] >= 0:
			widths[i] = p.shared[i]
		case c.auto:
			cell := ""
			if i < len(row) {
				cell = row[i]
			}
			widths[i] = runesWidth(parseSGR(cell, sketchStyle{}))
			used += widths[i]
		default:
			fills++
		}
	}
	left := max(p.total-used-p.gaps, 0)
	for i, c := range p.columns {
		if !p.visible[i] || p.shared[i] >= 0 || c.auto {
			continue
		}
		share := left / fills
		if fills == 1 {
			share = left
		}
		widths[i] = max(share, c.minWidth)
		left = max(left-widths[i], 0)
		fills--
	}
	return widths
}

// paletteColour resolves a colour of a column or a cell: a name of the
// design's palette (@Name) through the palette the screen hands its parts, any
// other text as it is.
func paletteColour(p props, c string) string {
	c = strings.TrimSpace(c)
	name, isName := strings.CutPrefix(c, "@")
	if !isName {
		return c
	}
	for _, entry := range strings.Split(p.str("theme.palette"), ";") {
		if key, value, ok := strings.Cut(entry, "="); ok && key == name {
			return value
		}
	}
	return ""
}

// cellStyleOf reads the style a program gave one cell: "fg|bg|flags" with the
// flags b (bold), d (dim) and r (reverse).
func cellStyleOf(p props, code string, base sketchStyle) sketchStyle {
	if code == "" || code == "-" {
		return base
	}
	parts := strings.Split(code, "|")
	if len(parts) > 2 && strings.Contains(parts[2], "p") {
		// Plain: the terminal's own colours, whatever the column says.
		base.Fg, base.Bg = "", ""
	}
	if len(parts) > 0 && parts[0] != "" {
		base.Fg = paletteColour(p, parts[0])
	}
	if len(parts) > 1 && parts[1] != "" {
		base.Bg = paletteColour(p, parts[1])
	}
	if len(parts) > 2 {
		base.Bold = base.Bold || strings.Contains(parts[2], "b")
		base.Dim = base.Dim || strings.Contains(parts[2], "d")
		base.Reverse = base.Reverse || strings.Contains(parts[2], "r")
	}
	return base
}

// drawRows draws the Rows component: one line per row, the cells laid out in
// the columns the design names, and each row in the style it was given:
// "selected" inverts the row in the accent colour (or, when a column has a
// selbg, only gives those columns their background), "dim" mutes it and
// "accent" colours its text. A column can set its own colours, cut with an
// ellipsis, or draw a colour as a block or as glyphs; a program can restyle a
// cell, and the text of a cell may carry SGR colours.
func drawRows(g *sketchGrid, p props) {
	accent := p.str("color")
	columns := rowsColumns(p.str("columns"))
	styles := split(p.str("styles"), ",")
	cellStyles := rowsCells(p.str("cellstyles"))
	anySelBg := false
	for _, c := range columns {
		anySelBg = anySelBg || c.selBg != "" || c.selFg != "" || c.selBold
	}
	allRows := rowsCells(p.str("rows"))
	gap := min(max(p.integer("gap", 0), 0), 8)
	plan := newRowsPlan(columns, allRows, g.W, gap)
	lastVisible := -1
	for i := range columns {
		if plan.visible[i] {
			lastVisible = i
		}
	}
	// A pane without focus shows its selected row as accent text, not inverted.
	blurred := p.str("focused") == "false"
	blurAccent := orColour(p.str("blurColor"), accent)
	for y, row := range allRows {
		if y >= g.H {
			break
		}
		base := sketchStyle{}
		isSelected := false
		if y < len(styles) {
			switch styles[y] {
			case "selected":
				isSelected = true
				if blurred {
					base = sketchStyle{Fg: blurAccent}
				} else if !anySelBg {
					base = sketchSelected(accent)
					g.Fill(sketchRect{X: 0, Y: y, W: g.W, H: 1}, ' ', base)
				}
			case "dim":
				base = sketchDim
			case "accent":
				base = sketchStyle{Fg: accent}
			}
		}
		x := 0
		widths := plan.row(row)
		for i, col := range columns {
			if !plan.visible[i] {
				continue
			}
			w := min(widths[i], max(g.W-x, 0))
			cell := ""
			if i < len(row) {
				cell = row[i]
			}
			style := base
			if col.fg != "" {
				style.Fg = paletteColour(p, col.fg)
			}
			if col.bg != "" {
				style.Bg = paletteColour(p, col.bg)
			}
			if isSelected && blurred {
				style.Fg = blurAccent
			} else {
				if isSelected && col.selBg != "" {
					style.Bg = paletteColour(p, col.selBg)
				}
				if isSelected && col.selFg != "" {
					style.Fg = paletteColour(p, col.selFg)
				}
				if isSelected && col.selBold {
					style.Bold = true
				}
			}
			if y < len(cellStyles) && i < len(cellStyles[y]) {
				style = cellStyleOf(p, cellStyles[y][i], style)
			}
			switch {
			case w <= 0:
			case col.colour && cell != "":
				g.Fill(sketchRect{X: x, Y: y, W: w, H: 1}, ' ', sketchStyle{Bg: cell})
			case col.glyph && cell != "":
				g.Text(x, y, strings.Repeat("█", w), sketchStyle{Fg: cell}, w)
			case col.colour || col.glyph:
				g.Text(x, y, "·", sketchDim, w)
			default:
				if style.Bg != "" && (!isSelected || !col.fit || col.selBg == "") {
					g.Fill(sketchRect{X: x, Y: y, W: w, H: 1}, ' ', sketchStyle{Bg: style.Bg})
				}
				at := x
				for _, run := range cutRuns(parseSGR(cell, style), w, col.ellipsis) {
					at += g.Text(at, y, run.text, run.style, 0)
				}
			}
			x += w
			if i < lastVisible {
				x += gap
			}
		}
	}
}

// keybarHints reads the hints property: "key:label" items separated by commas;
// the key ends at the first colon.
func keybarHints(spec string) [][2]string {
	var out [][2]string
	for _, item := range rowsSplit(spec, ",", true) {
		key, label, ok := strings.Cut(item, ":")
		if !ok {
			key, label = item, ""
		}
		if strings.TrimSpace(key) == "" && strings.TrimSpace(label) == "" {
			continue
		}
		out = append(out, [2]string{key, label})
	}
	return out
}

// drawKeybar draws the key bar: a space, then for each hint the key in bold
// in the key colour, a space and the label in the label colour, the hints
// separated by gap spaces, cut at the width.
func drawKeybar(g *sketchGrid, p props) {
	keyStyle := sketchStyle{Fg: p.str("keyColor"), Bold: true}
	labelStyle := sketchStyle{Fg: p.str("labelColor")}
	gap := min(max(p.integer("gap", 2), 0), 8)
	x := 1
	for i, h := range keybarHints(p.str("hints")) {
		if i > 0 {
			x += gap
		}
		if x >= g.W {
			break
		}
		x += g.Text(x, 0, h[0], keyStyle, g.W-x)
		if h[1] != "" && x < g.W {
			x += g.Text(x, 0, " ", sketchStyle{}, g.W-x)
			x += g.Text(x, 0, h[1], labelStyle, g.W-x)
		}
	}
}


// orColour is the colour when it is set, else the fallback.
func orColour(c, fallback string) string {
	if strings.TrimSpace(c) == "" {
		return fallback
	}
	return c
}

// drawProgressLine draws a progress bar followed by a text: barWidth cells of
// the full rune for the share done (rounded down) and of the empty rune for
// the rest, the gap, then the suffix. A barWidth of 0 takes what the suffix
// leaves of the width.
func drawProgressLine(g *sketchGrid, p props) {
	suffix := p.str("suffix")
	gap := min(max(p.integer("gap", 3), 0), 16)
	barWidth := p.integer("barWidth", 32)
	if barWidth <= 0 {
		barWidth = g.W - gap - sketchWidth(suffix)
	}
	barWidth = min(max(barWidth, 0), g.W)
	value := p.float("value", 0)
	value = min(max(value, 0), 100)
	done := int(value / 100 * float64(barWidth))
	full := firstRune(p.str("full"), '█')
	empty := firstRune(p.str("empty"), '░')
	fullStyle := sketchStyle{Fg: p.str("fullColor")}
	emptyStyle := sketchStyle{Fg: p.str("emptyColor")}
	for i := 0; i < barWidth; i++ {
		if i < done {
			g.Set(i, 0, sketchCell{Ch: full, Style: fullStyle})
		} else {
			g.Set(i, 0, sketchCell{Ch: empty, Style: emptyStyle})
		}
	}
	x := barWidth + gap
	if x < g.W {
		g.Text(x, 0, suffix, sketchStyle{Fg: p.str("suffixColor")}, g.W-x)
	}
}

func firstRune(s string, fallback rune) rune {
	if r := []rune(s); len(r) > 0 {
		return r[0]
	}
	return fallback
}

// slotPart is the view a program drew itself, cut to the rectangle of the
// design; the escape sequences of its colours pass through.
func slotPart(c placed, p props) *part {
	view := p.str("view")
	if view == "" {
		return staticPart("")
	}
	return staticPart(lipgloss.NewStyle().MaxWidth(c.W).MaxHeight(c.H).Render(view))
}
